+/// Remove `del` from the default graph and add `add`, in one commit.
+fn change(s: &Store, add: &[Triple], del: &[Triple]) {
+    let mut txn = s.write();
+    let mut labels = Default::default();
+    let q = |t: &Triple| {
+        Quad::new(

+            t.predicate.clone(),
