# job.sh, run in the named hc terminal m837-bench on atlas
#!/usr/bin/env bash
# Item 837 phase 2 on atlas: origin/main 9fe7dbd8 (base) against 051a5302 (p2), one job
# under ~/lanes/.heavy.lock, as item 113's matrix and item 938's large changes.
set -uo pipefail
L=$(cd "$(dirname "$0")" && pwd)
cd "$L" || exit 1
scope() { systemd-run --user --scope -q -p MemoryMax=13G -p MemorySwapMax=0 -p OOMPolicy=continue "$@"; }
log() { echo "[$(date +%T)] $*" >> "$L/job.log"; }
exec 9>"$HOME/lanes/.heavy.lock"
log "waiting for the heavy lock"
flock 9
log "lock taken load=$(cut -d' ' -f1 /proc/loadavg)"
mkdir -p data
if [ ! -d data/db-tip ]; then
  log "generating and loading LUBM(100)"
  bin/lubm-base gen 100 data/src > data/gen.log 2>&1 || { log "gen failed"; exit 1; }
  scope bin/lubm-base load data/src 100 data/db-tip > data/load.log 2>&1 || { log "load failed"; exit 1; }
  rm -rf data/src
fi
log "waiting for bin/ready"
until [ -e bin/ready ]; do sleep 20; done
log "matrix p2=$(sha256sum bin/lubm-p2 | cut -c1-16) base=$(sha256sum bin/lubm-base | cut -c1-16)"
ARMS="base p2" PIN1=2-5 TAG=atlas ./ab.sh
log "large changes read back from the dataset (item 938)"
mkdir -p runs-938
for round in 1 2; do
  for arm in base p2; do
    rm -rf data/run-938.db && cp -r data/db-tip data/run-938.db
    RAYON_NUM_THREADS=20 scope bin/lubm-$arm keep "$L/data/run-938.db" owl-rl 100000 \
      > "runs-938/$arm-r$round.out" 2> "runs-938/$arm-r$round.err"
    log "938 $arm r$round rc=$?"
  done
done
rm -rf data/run-938.db
log "job-finished load=$(cut -d' ' -f1 /proc/loadavg)"

# ab.sh
#!/usr/bin/env bash
# Interleaved rounds of `lubm churn` for two builds, each on a fresh copy of the dataset,
# after item 113's ab.sh. Env: ARMS PROFILES MODES ROUNDS SIZES BATCHES THREADS PIN1 MEM TAG
set -uo pipefail
L=$(cd "$(dirname "$0")" && pwd)
ARMS=${ARMS:-"base p2"}
PROFILES=${PROFILES:-"rdfs owl-rl"}
MODES=${MODES:-"t1 p16"}
ROUNDS=${ROUNDS:-3}
SIZES=${SIZES:-100,1000,10000}
BATCHES=${BATCHES:-40,20,6}
THREADS=${THREADS:-16}
PIN1=${PIN1:-}
MEM=${MEM:-13G}
R=$L/runs${TAG:+-$TAG}
mkdir -p "$R" && cd "$L/data" || exit 1
last=${SIZES##*,}
read -r -a arms <<< "$ARMS"
n=${#arms[@]}
for round in $(seq 1 "$ROUNDS"); do
  for profile in $PROFILES; do
    for mode in $MODES; do
      for i in $(seq 0 $((n - 1))); do
        arm=${arms[$(((i + round - 1) % n))]}
        name=$arm-$profile-$mode-r$round
        if grep -q "k=$last rounds=.* revert-again first_us" "$R/$name.out" 2> /dev/null; then
          continue
        fi
        run=run-ab${TAG:+-$TAG}.db
        rm -rf "$run" && cp -r db-tip "$run"
        threads=$THREADS pool=0 pin=
        case $mode in
          t1) threads=1 pin=$PIN1 ;;
          p16) pool=1 ;;
        esac
        env RUST_LOG=sparkles_reasoner::timing=trace LUBM_POOL=$pool RAYON_NUM_THREADS=$threads \
          systemd-run --user --scope -q -p "MemoryMax=$MEM" -p MemorySwapMax=0 -p OOMPolicy=continue \
          ${pin:+taskset -c "$pin"} "$L/bin/lubm-$arm" churn "$L/data/$run" "$profile" "$SIZES" 100 "$BATCHES" \
          > "$R/$name.out" 2> "$R/$name.err"
        echo "[$(date +%T)] $name rc=$? load=$(cut -d' ' -f1 /proc/loadavg)" >> "$R/progress.log"
        rm -rf "$run"
      done
    done
  done
done
echo "[$(date +%T)] ab-finished" >> "$R/progress.log"
