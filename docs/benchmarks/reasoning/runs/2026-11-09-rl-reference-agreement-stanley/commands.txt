python3 -I scripts/rl-reference/w3c.py all.rdf WORK/w3c

# bench.sh ran compare.sh on the base data (lubm-1-aligned.json, lubm-1-core.json,
# owl2bench-1-aligned.json, owl2bench-1-core.json) and compared the closures that
# incremental maintenance left after inserting and deleting its change
# (*-after-insert.json, lubm-1-after-delete.json). Its deletion run on OWL2Bench(1) ran out
# of the 8G limit, and was repeated by hand on a copy of the materialized store with the
# change deleted (owl2bench-1-delete.txt):
sparkles update --loc del.db --update delete.ru
systemd-run --user --scope -q -p MemoryMax=16G -p MemorySwapMax=0 timeout 600 \
  sparkles infer --loc del.db --profile owl-rl --threads 16

# owl2bench-1-runs.jsonl is bench.sh's record of that OWL2Bench(1) round, and
# owl2bench-1-full-t16.time and owl2bench-1-delete-8g.time are GNU time's reports of the
# full run on 16 threads (580,428 KiB, or 566 MiB, at most) and of the deletion run that the
# 8G scope ended. This bench.sh predates the later harness, so its "seconds" span the scope
# and its "oom" follows exit status 137. owl2bench-1-delete-oom.txt holds the kernel's
# records of both deletion runs. The harness run was killed at its cgroup's limit with
# 8.0 GiB of anonymous memory, and the run by hand was killed by the host's out-of-memory
# killer, below its 16G scope limit, with 14.1 GiB after 136 s.
