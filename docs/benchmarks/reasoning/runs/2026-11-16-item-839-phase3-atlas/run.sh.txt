#!/usr/bin/env bash
# Item 839 Phase 3 gate on atlas, under ~/lanes/.heavy.lock: base dd964cf0 against the lane
# tip. LUBM(100) incremental runs of `lubm keep` (kept and read back) for rdfs and owl-rl
# with changes of 100 and 10,000 triples, `examples/incremental.rs` on 1M generated triples
# for rdfs and owl-rl, and full `lubm infer` runs at 1 and 20 threads, ROUNDS interleaved
# rounds with alternating arm order.
set -uo pipefail
D=$HOME/lanes/datalog-839
TIME=/run/current-system/sw/bin/time
ROUNDS=${ROUNDS:-3}
cd "$D" || exit 1
mkdir -p logs3 data scratch
log() { echo "[$(date +%T)] $*" | tee -a progress3.log; }
scope() { systemd-run --user --scope -q -p MemoryMax=13G -p MemorySwapMax=0 -p OOMPolicy=continue "$@"; }
B=$D/bin3/lubm-base
if [ ! -d data/lubm100.db ]; then
  log "generating and loading LUBM(100)"
  "$B" gen 100 data/lubm100 > data/gen.log 2>&1 || { log "gen failed"; exit 1; }
  scope "$B" load data/lubm100 100 data/lubm100.db > data/load.log 2>&1 || { log "load failed"; exit 1; }
fi
timed() {
  local name=$1 bin=$2 cpus=$3 threads=$4; shift 4
  local rc=0
  RAYON_NUM_THREADS=$threads scope "$TIME" -v taskset -c "$cpus" "$bin" "$@" > "logs3/$name.out" 2> "logs3/$name.err" || rc=$?
  local rss
  rss=$(sed -n 's/.*Maximum resident set size (kbytes): //p' "logs3/$name.err")
  echo "$name rc=$rc rss_kib=$rss load=$(cut -d' ' -f1 /proc/loadavg)" | tee -a results3.txt progress3.log
}
for r in $(seq 1 "$ROUNDS"); do
  if [ $((r % 2)) = 1 ]; then arms="base p3"; else arms="p3 base"; fi
  for profile in rdfs owl-rl; do
    for k in 100 10000; do
      for arm in $arms; do
        rm -rf scratch/db; cp -r data/lubm100.db scratch/db
        timed "keep-$profile-$k-$arm-r$r" "$D/bin3/lubm-$arm" 0-19 20 keep scratch/db "$profile" "$k"
      done
    done
    for arm in $arms; do
      rm -rf scratch/inc
      timed "incremental-$profile-$arm-r$r" "$D/bin3/incremental-$arm" 0-19 20 scratch/inc 1000000 "$profile"
    done
    for t in 1 20; do
      cpus=2; [ "$t" = 20 ] && cpus=0-19
      for arm in $arms; do
        timed "infer-$profile-$arm-t$t-r$r" "$D/bin3/lubm-$arm" "$cpus" "$t" infer data/lubm100.db "$profile"
      done
    done
  done
done
rm -rf scratch
log "DONE"

  done

#!/usr/bin/env bash
# Item 839 Phase 3 bisect on atlas, under ~/lanes/.heavy.lock: one-thread `lubm infer` under
# owl-rl on OWL2Bench(5) and LUBM(100), CPU 2, for base dd964cf0, p3 6e03b4d3 (bin3 and a
# rebuild, p3re), va (p3 with the RelTuple arm of builtins::eval as in base) and vb (p3
# with run_program/run_from's cancel checks and set_deadline calls as in base).

TIME=/run/current-system/sw/bin/time
ROUNDS=${ROUNDS:-3}

timed() {
  local name=$1 arm=$2; shift 2
  local rc=0
  RAYON_NUM_THREADS=1 SPARKLES_MAX_INFERRED=200000000 scope "$TIME" -v taskset -c 2 "$D/bin3/lubm-$arm" "$@" > "logs3d/$name.out" 2> "logs3d/$name.err" || rc=$?
  local rss
  rss=$(sed -n 's/.*Maximum resident set size (kbytes): //p' "logs3d/$name.err")
  echo "$name rc=$rc rss_kib=$rss load=$(cut -d' ' -f1 /proc/loadavg)" | tee -a results3d.txt progress3d.log

log "start $ROUNDS rounds"
all=(base p3 p3re va vb)
for r in $(seq 1 "$ROUNDS"); do
  arms=("${all[@]:$((r % 5))}" "${all[@]:0:$((r % 5))}")
  for arm in "${arms[@]}"; do
    timed "infer-owl2bench5-$arm-t1-r$r" "$arm" infer data/owl2bench5.db owl-rl
  done
  for arm in "${arms[@]}"; do
    timed "infer-owl-rl-$arm-t1-r$r" "$arm" infer data/lubm100.db owl-rl
  done

# atlas3f.sh

#!/usr/bin/env bash
# Item 839 Phase 3: user-space instruction and cycle counts of one-thread `lubm infer` under
# owl-rl on LUBM(100), basere (dd964cf0) against p3re (6e03b4d3), built in one worktree,
# CPU 2, two runs each, interleaved, under ~/lanes/.heavy.lock.
set -uo pipefail
D=$HOME/lanes/datalog-839
PERF=/nix/store/ciwqn0d1ybrvzw9sdlklfhg58njv3inc-perf-linux-7.3.3/bin/perf
cd "$D" || exit 1
mkdir -p logs3f
for r in 1 2; do
  arms="basere p3re"; [ "$r" = 2 ] && arms="p3re basere"
  for arm in $arms; do
    RAYON_NUM_THREADS=1 SPARKLES_MAX_INFERRED=200000000 systemd-run --user --scope -q -p MemoryMax=13G -p MemorySwapMax=0 -p OOMPolicy=continue \
      "$PERF" stat -x, -o "logs3f/stat-$arm-r$r.csv" -e instructions:u,cycles:u,branch-misses:u,L1-icache-load-misses:u,iTLB-load-misses:u \
      taskset -c 2 "$D/bin3/lubm-$arm" infer data/lubm100.db owl-rl > "logs3f/infer-$arm-r$r.out" 2>&1
    echo "$arm r$r rc=$? $(grep -oE 'ms=[0-9]+' logs3f/infer-$arm-r$r.out)" | tee -a progress3f.log
  done
done
echo DONE | tee -a progress3f.log
