#!/usr/bin/env python3
"""Item 829 script rerun on forge: three rounds of the text, geo and vector scripts.

    analyze_rounds.py OUT_DIR > summary.md   (OUT_DIR holds r1, r2, r3)

Each round runs every arm once, A,B,B,A in rounds 1 and 3 (base, tip, tip2, base2) and
B,A,A,B in round 2 (tip, base, base2, tip2). A round gives two pairs, tip/base and
tip2/base2, so three rounds give six pairs. Per query the ratio of each pair is the median
of the tip requests over the median of the base requests (client wall time). The
same-binary ratios base2/base and tip2/tip of each round give the spread.

The rule is the first pass's rule extended to six pairs (the plan's 5% on wall time and
AGENTS.md's noise rule for non-Intel A/Bs, under which a difference that is not above 1
in every pair is noise). A query is slower when every pair is above 1. The first
pass's written gate rule is "every pair >= 1.05" (min pair); the pooled ratio (median of
all tip requests over all base requests) and the per-round ratios are reported beside it,
and a slower query whose pooled ratio is 1.05 or more is flagged for the coordinator.
Answer fingerprints must agree across every arm, mode and thread count of a round.
"""

import json
import statistics as st
import sys
from pathlib import Path

SCRIPTS = ("text", "geo", "vector")
ARMS = ("base", "tip", "tip2", "base2")


def ab_times(d, script):
    t = {}
    if script == "vector":
        h = json.loads((d / "hyperfine.json").read_text())
        for r in h["results"]:
            t[r["name"]] = [m["time_wall_clock"]["value"] for m in r["measurements"]]
        return t
    for f in sorted(d.glob(f"{script}-*.json")):
        if f.name in (f"{script}-answers.json", f"{script}-status.json", "text-hits.json"):
            continue
        j = json.loads(f.read_text())
        rs = [r for r in j.get("results", []) if r.get("command") == "sparkles"]
        if rs:
            t[f.stem[len(script) + 1:]] = rs[0]["times"]
    return t


def answers(d, script):
    if script == "vector":
        j = json.loads((d / "fingerprint.json").read_text())
        j.pop("execution", None)
        return json.dumps(j, sort_keys=True)
    j = json.loads((d / f"{script}-answers.json").read_text())
    return json.dumps(j, sort_keys=True)


def main():
    out = Path(sys.argv[1])
    rounds = sorted(p for p in out.glob("r*") if p.is_dir())
    res = {}
    bad = []
    for script in SCRIPTS:
        ref = None
        for r in rounds:
            for threads in ("t12", "t1"):
                for mode in ("eager", "streaming"):
                    for a in ARMS:
                        d = r / "results" / f"{script}-{a}-{mode}-{threads}"
                        fp = answers(d, script)
                        if ref is None:
                            ref = fp
                        elif fp != ref:
                            bad.append(f"{r.name}/{script}-{a}-{mode}-{threads}")
    print("# Item 829 script rerun on forge, base bccc5f7b against tip 9df856f3, 3 rounds\n")
    print(f"Answer fingerprints differing from the first arm: {bad or 'none'}\n")
    for script in SCRIPTS:
        for threads in ("t12", "t1"):
            for mode in ("eager", "streaming"):
                per = {}
                for r in rounds:
                    per[r.name] = {a: ab_times(r / "results" / f"{script}-{a}-{mode}-{threads}", script)
                                   for a in ARMS}
                print(f"\n### {script} {mode} {threads}\n")
                print("| query | base ms | tip ms | pooled | pairs (r1 r1' r2 r2' r3 r3') | per round | min pair | same-binary max | verdict |")
                print("|---|--:|--:|--:|---|---|--:|--:|---|")
                qs = sorted(set.intersection(*(set(v) for rr in per.values() for v in rr.values())))
                for q in qs:
                    pairs, sbs, base_all, tip_all, rr = [], [], [], [], []
                    for rn, arms in per.items():
                        x = {a: arms[a][q] for a in ARMS}
                        pairs += [st.median(x["tip"]) / st.median(x["base"]),
                                  st.median(x["tip2"]) / st.median(x["base2"])]
                        sbs += [st.median(x["base2"]) / st.median(x["base"]),
                                st.median(x["tip2"]) / st.median(x["tip"])]
                        rr.append(st.median(x["tip"] + x["tip2"]) / st.median(x["base"] + x["base2"]))
                        base_all += x["base"] + x["base2"]
                        tip_all += x["tip"] + x["tip2"]
                    pooled = st.median(tip_all) / st.median(base_all)
                    spread = max(max(s, 1 / s) for s in sbs) - 1
                    if all(p > 1 for p in pairs):
                        vd = "slower in every pair"
                        if pooled >= 1.05:
                            vd += ", pooled >= 1.05" + (" (12-thread gate question)" if threads == "t12" else " (one thread, 1162)")
                        if min(pairs) >= 1.05:
                            vd += " (every pair >= 1.05)"
                    elif all(p < 1 for p in pairs):
                        vd = "faster in every pair"
                    else:
                        vd = "noise"
                    res[f"{script}/{mode}/{threads}/{q}"] = dict(
                        base_ms=st.median(base_all) * 1e3, tip_ms=st.median(tip_all) * 1e3,
                        pooled=pooled, pairs=pairs, per_round=rr, same_binary=sbs, verdict=vd)
                    print(f"| {q} | {st.median(base_all)*1e3:.2f} | {st.median(tip_all)*1e3:.2f} | "
                          f"{pooled:.3f} | {' '.join(f'{p:.3f}' for p in pairs)} | {' '.join(f'{p:.3f}' for p in rr)} | {min(pairs):.3f} | "
                          f"{spread*100:.1f}% | {vd} |")
    (out.parent / "rounds-analysis.json").write_text(json.dumps(dict(answers_differ=bad, ab=res), indent=1))


main()
