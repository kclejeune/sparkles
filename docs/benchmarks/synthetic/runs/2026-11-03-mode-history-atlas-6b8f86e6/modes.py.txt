#!/usr/bin/env python3
"""Per server process median exec time of each query, by variant, and the B/A ratio of
pooled medians. usage: modes.py samples.tsv [threshold]"""
import csv, statistics as st, collections, sys
th = float(sys.argv[2]) if len(sys.argv) > 2 else None
d = collections.defaultdict(list)
for r in csv.DictReader(open(sys.argv[1]), delimiter='\t'):
    if r['exec_ms'] == 'error': continue
    d[(r['query'], int(r['epoch']), r['variant'])].append(float(r['exec_ms']))
qs = sorted({k[0] for k in d})
for q in qs:
    out = {'A': [], 'B': []}; allv = {'A': [], 'B': []}
    for (qq, e, v), xs in sorted(d.items(), key=lambda kv: kv[0][1]):
        if qq != q: continue
        out[v].append(st.median(xs)); allv[v] += xs
    ma, mb = st.median(allv['A']), st.median(allv['B'])
    line = f"{q}: A {ma:.3f} B {mb:.3f} B/A {mb/ma:.3f} | per-proc A " + ' '.join(f'{x:.2f}' for x in out['A']) + ' | B ' + ' '.join(f'{x:.2f}' for x in out['B'])
    if th:
        line += f" | slow A {sum(x>th for x in out['A'])}/{len(out['A'])} B {sum(x>th for x in out['B'])}/{len(out['B'])}"
    print(line)
