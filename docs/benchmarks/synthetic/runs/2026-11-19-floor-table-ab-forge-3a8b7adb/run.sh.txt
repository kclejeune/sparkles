#!/usr/bin/env bash
# final-ab-forge-s11.sh NEW: session 11 rerun of f1-f4 as g1-g4 for the shared floor table,
# plus g5 (streaming) and g6 (eager) at 1 thread against the per-worker maps 174de1f8.
# Builds NEW, c0948014 and the direct probe 971717d2 on forge (cached -forge entries are
# reused), stages g1-g4 and queues them on forge's heavy lock as unit floor-cache-1069-g-forge.
# Driver done when its output ends "exit 0". Results done when forge
# ~/lanes/floor-cache-1069/status-g ends with DONE (FAILED lines name failed runs).
set -euo pipefail
cd "$(dirname "$0")"
G=$REPO
NEW=$(git -C $G rev-parse "$1^{commit}")
C=$HOME/.cache/sparkles-nsc/bin
MAIN=c0948014705b4bb40aaa7fe4c86c3c671444b7e9 OLD=75a70587acbbbc026f91d279dbabaad1003ab278 BASE=7e272f4dabc6b1b3e5f001690573d90ba07afcba
PR=$(git -C $G rev-parse "971717d2^{commit}")
bash forge-build.sh "$NEW" "$MAIN" "$PR"
for s in "$NEW" "$MAIN" "$PR"; do
  grep -qx 'build host: forge' "$C/$s-forge/build.txt" || { echo "$s-forge is not a forge build"; exit 1; }
done
export HOST=forge EXTRA_Q=$PWD/queries-extra SUFFIX_A=-forge SUFFIX_B=-forge
SQ="str-pair-lt str-pair-names str-pair-colleague str-filter-two"
TQ="str-filter-two lang-filter distinct-join employee-docs"
WQ="str-pair-lt str-pair-names str-pair-colleague str-filter-two bind-two order-lcase"
bash stage.sh g1-streaming $MAIN $NEW 100000 5 10 streaming - "$SQ"
bash stage.sh g2-vs-direct $PR $NEW 100000 5 10 eager - "$SQ"
RAYON=12 bash stage.sh g3-t12-streaming $MAIN $NEW 100000 5 10 streaming - "$WQ"
RAYON=12 bash stage.sh g4-t12-eager $MAIN $NEW 100000 5 10 eager - "$WQ"
MAPS=$(git -C $G rev-parse "174de1f8^{commit}")
grep -qx "build host: forge" "$C/$MAPS-forge/build.txt" || { echo "$MAPS-forge is not a forge build"; exit 1; }
MQ="str-pair-names str-pair-colleague str-pair-lt"
bash stage.sh g5-vs-maps-streaming $MAPS $NEW 100000 5 10 streaming - "$MQ"
bash stage.sh g6-vs-maps-eager $MAPS $NEW 100000 5 10 eager - "$MQ"
RUNS="g1-streaming g2-vs-direct g3-t12-streaming g4-t12-eager g5-vs-maps-streaming g6-vs-maps-eager"
H=forge U=floor-cache-1069-g-forge
ssh $H "systemctl --user is-active -q $U" && { echo "$U already runs on $H"; exit 1; }
ssh $H 'mkdir -p ~/lanes/floor-cache-1069 && rm -f ~/lanes/floor-cache-1069/status-g'
for n in $RUNS; do rsync -a --delete "$PWD/$n/stage/" "$H:lanes/floor-cache-1069/$n/"; done
ssh $H "systemd-run --user --unit=$U --collect --working-directory=~/lanes/floor-cache-1069 -p MemoryMax=12G \
  flock ~/lanes/.heavy.lock bash -c 'for n in $RUNS; do (cd \$n && bash nsc-bench-remote.sh) > \$n.log 2>&1 || echo FAILED \$n >> status-g; echo ended \$n >> status-g; done; echo DONE >> status-g'"
echo "queued on forge as $U"
