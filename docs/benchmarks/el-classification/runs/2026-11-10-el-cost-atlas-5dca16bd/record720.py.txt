#!/usr/bin/env python3
"""Writes the record.json of the atlas measurements of items 720, 721 and 722 from its run directories."""
import json, statistics as st, sys
from collections import defaultdict
from pathlib import Path
R = Path(sys.argv[1])
med = st.median
def rows(p):
    return [json.loads(l) for l in (p / "runs.jsonl").read_text().splitlines() if l.strip()]
def cold(part):
    out = {}
    for d in sorted((R / part).glob("cold-*")):
        t = int(dict(x.split("=") for x in (d / "env.txt").read_text().split()[:2] if "=" in x).get("threads", "4"))
        g = defaultdict(list)
        for r in rows(d):
            k = "cold" if r["repeat"] == 1 else "warm"
            g[(r["file"], r["side"], k)].append(r["coldMs"] if k == "cold" else r["warmMs"])
        out[f"threads_{t}"] = {f: {s: {k: round(med(v), 3) for (f2, s2, k), v in g.items() if f2 == f and s2 == s}
                                   for s in sorted({s for _, s, _ in g})} for f in sorted({f for f, _, _ in g})}
        out[f"threads_{t}_rounds"] = {f"{f} {s} {k}": [round(x, 3) for x in v] for (f, s, k), v in sorted(g.items())}
    return out
def ab(part):
    out = {}
    for d in sorted((R / part).glob("out-*")):
        g = defaultdict(list)
        for r in rows(d):
            if r.get("failed"):
                g[(r["file"], r["threads"], r["side"])].append(None); continue
            g[(r["file"], r["threads"], r["side"])].append(r)
        res = {}
        for (f, t, s), v in sorted(g.items()):
            ok = [x for x in v if x]
            res[f"{f} t{t} {s}"] = {"classify_median_ms": round(med(x["classifyMedianMs"] for x in ok), 2),
                                    "saturation_ms": round(med(x["saturationMs"] for x in ok), 2),
                                    "index_ms": round(med(x["indexMs"] for x in ok), 2),
                                    "rounds_ms": [round(x["classifyMedianMs"], 2) for x in ok],
                                    "failed": len(v) - len(ok)}
        out[d.name] = {"results": res, "taxonomy_sha256": (d / "taxonomy.sha256").read_text().split("\n")}
    return out
def perf(part):
    out = {}
    for f in sorted((R / part / "perf").glob("*.csv")):
        n, side, r, _ = f.name.split(".")
        vals = {}
        for l in f.read_text().splitlines():
            p = l.split(",")
            if len(p) > 3 and p[0][:1].isdigit():
                vals[p[2].split("/")[1]] = int(float(p[0]))
        out[f"{n} {side} {r}"] = vals
    return out
def mem(part):
    out = {}
    for f in sorted((R / part / "mem").glob("*.json")):
        j = json.loads(f.read_text())
        out[f.stem] = {"classify_s": round(j["classifyMs"][0] / 1000, 2), "peak_rss_gib": round(j["peakRssKib"] / 2**20, 2),
                       "peak_estimate_gib": round(j["peakEstimatedBytes"] / 2**30, 2), "conclusions": j["conclusions"]}
    return out
