+/// Diagnostics of the lane horn-cost, never committed: counters of the work of Tier D
+/// that `SPARKLES_DL_DIAG` prints at every doubling of the derived conclusions and when
+/// the run ends.
+pub mod diag {
+    use std::sync::OnceLock;
+    use std::sync::atomic::{AtomicU64, Ordering::Relaxed};
+    pub const NAMES: [&str; 10] = [
+        "derive", "foldChoice", "foldNew", "addOk", "addSub", "processed", "predProd", "pruneIn", "foldEmit", "maxProd",
+    ];
+    pub static C: [AtomicU64; 10] = [const { AtomicU64::new(0) }; 10];
+    static NEXT: AtomicU64 = AtomicU64::new(1 << 16);
+    fn start() -> &'static std::time::Instant {
+        static T: OnceLock<std::time::Instant> = OnceLock::new();
+        T.get_or_init(std::time::Instant::now)
+    }
+    #[inline]
+    pub fn on() -> bool {
+        static ON: OnceLock<bool> = OnceLock::new();
+        *ON.get_or_init(|| std::env::var_os("SPARKLES_DL_DIAG").is_some())
+    }
+    #[inline]
+    pub fn add(i: usize, n: u64) {
+        if on() {
+            C[i].fetch_add(n, Relaxed);
+        }
+    }
+    pub fn line(tag: &str, work: usize) {
+        let mut s = format!("DLDIAG {tag} work={work} ms={}", start().elapsed().as_millis());
+        for (i, n) in NAMES.iter().enumerate() {
+            s.push_str(&format!(" {n}={}", C[i].load(Relaxed)));
+        }
+        eprintln!("{s}");
+    }
+    #[inline]
+    pub fn at(work: usize) {
+        if !on() {
+            return;
+        }
+        let _ = start();
+        let next = NEXT.load(Relaxed);
+        if work as u64 >= next
+            && NEXT
+                .compare_exchange(next, next * 2, Relaxed, Relaxed)
+                .is_ok()
+        {
+            line("at", work);
+        }
+    }
+    pub fn reset() {
+        if on() {
+            for c in &C {
+                c.store(0, Relaxed);
+            }
+            NEXT.store(1 << 16, Relaxed);
+        }
+    }
