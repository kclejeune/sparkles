#!/usr/bin/env bash
# dsweep.sh BIN TAG [args]: BIN on one thread over the 96 disjunctive dev files, four at a time,
# with SPARKLES_DL_DIAG; per file out/dsweep-TAG/NAME.{json,err}, a summary line in
# out/dsweep-TAG.txt.
L=/home/kclejeune/repositories/github.com/kclejeune/sparkles/.artifacts/worktrees/horn-cost/.artifacts/lanes/horn-cost
. $L/files.sh
bin=$1; tag=$2; shift 2
O=$L/out/dsweep-$tag; mkdir -p $O
one() {
  n=$1; slot=$2
  f=$(ls $DEV/$n.* | head -1)
  SPARKLES_DL_DIAG=1 timeout 400 systemd-run --user --scope -q -p MemoryMax=8G -p MemorySwapMax=1G -p OOMPolicy=continue \
    taskset -c $((12 + slot)) $L/bin/$bin --threads 1 --taxonomy $O/$n.tax $EXTRA $f > $O/$n.json 2> $O/$n.err
  st=$?
  sha=$(LC_ALL=C sort $O/$n.tax 2>/dev/null | sha256sum | cut -c1-16); rm -f $O/$n.tax
  echo "$n st=$st tax=$sha $(grep '^{' $O/$n.json | jq -c '[.classifyMs[0],.tier,.dlAttemptOver,.dlOverBudget,.dlWork,.dlContexts,.dlStored,.dlTests,.conclusions,.estimatedBytes,.peakEstimatedBytes]')" >> $L/out/dsweep-$tag.txt
}
export -f one; export L DEV O bin tag; export EXTRA="$*"
i=0
while read -r n; do echo "$n $((i % 4))"; i=$((i+1)); done < $L/disjunctive.txt |
  xargs -P 4 -L 1 bash -c 'one $0 $1' 2>/dev/null
echo SWEEP-DONE >> $L/out/dsweep-$tag.txt
