def lines(p):
    return [l for l in (R / p).read_text().splitlines() if l.strip()]
def env(d):
    e = {}
    for l in lines(f"{d}/env.txt"):
        if l.startswith("binary_sha256 ") or l.startswith("data_sha256 "):
            k, v = l.split(" ", 1)[1].split("=", 1); e.setdefault(l.split(" ")[0], {})[k] = v
        elif l.startswith("rounds="):
            m = re.match(r"rounds=(\d+) threads=([\d ]+?) repeat=(\d+) pin1=(\S+) pinn=(\S+)", l)
            e.update(rounds=int(m.group(1)), threads=[int(x) for x in m.group(2).split()], pin1=m.group(4), pinn=m.group(5))
        else:
            for part in l.split(" "):
                if "=" in part:
                    k, v = part.split("=", 1); e.setdefault(k, v)
    return e
r2 = lambda x: round(x, 2)
def gate(d="gate"):
    g = defaultdict(list)
    for l in lines(f"{d}/runs.jsonl"):
        r = json.loads(l); g[(r["file"], r["threads"], r["side"])].append(r)
    out = []
    for (f, t, s), rs in g.items():
        rs.sort(key=lambda r: r["round"])
        meds = [r2(r["classifyMedianMs"]) for r in rs]
        med = lambda k: r2(st.median([r[k] for r in rs]))
        out.append({"ontology": f, "threads": t, "side": s, "rounds": len(rs), "classifyMedianMs": meds,
                    "medianOfMediansMs": r2(st.median(meds)), "indexMs": med("indexMs"), "saturationMs": med("saturationMs"),
                    "taxonomyMs": med("taxonomyMs"), "peakRssKib": int(st.median([r["peakRssKib"] for r in rs])),
                    "conclusions": rs[0].get("conclusions"), "tier": rs[0].get("tier")})
    return out
def cold(d):
    g = defaultdict(lambda: defaultdict(list))
    for l in lines(f"{d}/runs.jsonl"):
        r = json.loads(l)
        assert r["failed"] is None, r
        k = (r["file"], r["side"])
        if r["repeat"] == 1:
            for x in ("coldMs", "wallMs", "readMs", "indexMs", "saturationMs", "taxonomyMs"):
                g[k][x].append(r[x])
        else:
            g[k]["warmMs"].append(r["warmMs"]); g[k]["tier"] = r["tier"]
    t = int(env(d)["threads"])
    return [{"ontology": f, "threads": t, "side": s, "rounds": len(v["coldMs"]), "coldMs": [r2(x) for x in v["coldMs"]],
             "coldMedianMs": r2(st.median(v["coldMs"])), "warmMedianMs": r2(st.median(v["warmMs"])),
             "wallMedianMs": r2(st.median(v["wallMs"])), "readMs": r2(st.median(v["readMs"])), "indexMs": r2(st.median(v["indexMs"])),
             "saturationMs": r2(st.median(v["saturationMs"])), "taxonomyMs": r2(st.median(v["taxonomyMs"])), "tier": v["tier"]}
            for (f, s), v in sorted(g.items())]
def checks(d="gate"):
    by = defaultdict(set)
    for l in lines(f"{d}/taxonomy.sha256"):
        h, n = l.split(); by[n.split(".")[0]].add(h)
    return {f: (sorted(h)[0] if len(h) == 1 else sorted(h)) for f, h in sorted(by.items())}

                    "peakEstimatedBytes": j["peakEstimatedBytes"], "estimatedBytes": j["estimatedBytes"], "conclusions": j["conclusions"],
                    "tier": j["tier"]})
    return out
