#!/usr/bin/env python3
"""summarize.py OUTDIR: medians over the rounds of timing.sh. One line per file, thread count
and arm: rounds, median of the processes' classifyMedianMs, its min and max, median index and
saturation ms, median peak RSS MiB, conclusions (min..max over rounds), ratio to base, and
whether every arm's taxonomy hash agrees. Fails loudly on a non-zero status or an empty hash."""
import json, sys, glob, os, statistics as st, collections
out = sys.argv[1]
hashes = collections.defaultdict(set); bad = []
for line in open(os.path.join(out, "log.txt")):
    p = line.split()
    if len(p) < 6: continue
    r, arm, name, t, status, h = p
    if status != "status=0" or h.startswith("e3b0c442"): bad.append(line.strip())
    hashes[name + (" el" if arm.endswith("el") else "")].add(h)
rows = collections.defaultdict(list)
for f in glob.glob(os.path.join(out, "r*", "*", "*.json")):
    arm = f.split(os.sep)[-2]; name, t = os.path.basename(f)[:-5].rsplit(".t", 1)
    try: d = json.load(open(f))
    except Exception as e: bad.append(f"{f}: {e}"); continue
    rows[(name, int(t), arm)].append(d)
order = ["base", "l0", "l1", "l2", "base-horn", "l0-horn", "l1-horn", "l2-horn", "base-el", "el"]
med = lambda k, ds: st.median(d[k] for d in ds)
print("file threads arm rounds classify_ms min max index_ms saturation_ms rss_mib conclusions vs_base tier")
for name, t in sorted({(n, t) for n, t, _ in rows}):
    for arm in order:
        ds = rows.get((name, t, arm))
        if not ds: continue
        b = rows.get((name, t, "base-horn" if arm.endswith("-horn") else "base-el" if arm.endswith("el") else "base"))
        m = med("classifyMedianMs", ds)
        ratio = f"{m / med('classifyMedianMs', b):.3f}" if b else "-"
        c = [d["conclusions"] for d in ds]
        print(name, t, arm, len(ds), f"{m:.0f}", f"{min(d['classifyMedianMs'] for d in ds):.0f}",
              f"{max(d['classifyMedianMs'] for d in ds):.0f}", f"{med('indexMs', ds):.0f}",
              f"{med('saturationMs', ds):.0f}", f"{med('peakRssKib', ds) / 1024:.0f}",
              f"{min(c)}..{max(c)}" if min(c) != max(c) else c[0], ratio, ds[0].get("tier"))
print("\nfile threads horn_ms/el_ms: base l0 l1 l2 (medians of the arms above)")
for name, t in sorted({(n, t) for n, t, a in rows if a == "el"}):
    el = {a: med("classifyMedianMs", rows[(name, t, a)]) for a in ("base-el", "el")}
    print(name, t, " ".join(f"{med('classifyMedianMs', rows[(name, t, a)]) / el['base-el' if a == 'base' else 'el']:.3f}"
                             for a in ("base", "l0", "l1", "l2") if rows.get((name, t, a))))
print("\nhashes that differ between arms:", {n: sorted(h) for n, h in hashes.items() if len(h) > 1} or "none")
print("bad lines:", bad or "none")
