#!/usr/bin/env bash
# timing1025.sh (roadmap item 1025, second half): bin/base is 48344bca and bin/tip the lane's
# committed tip, built by one command in one clean worktree on wally. Interleaved rounds, one
# process per arm, file and thread count; the caller holds ~/lanes/.heavy.lock. Arms: base, and
# the tip at SPARKLES_HORN_RULES 0, 1 and 2 (l0, l1, l2; 2 is the default). The four files of row
# 1025 also run the EL rules alone (--el-only) of both binaries (base-el, el), the denominator
# of the row's ratios. The two Uberon files run only with --dl-attempt-work 1, which ends the
# disjunctive attempt at once and times the Horn rules alone (suffix -horn), since the bounded
# disjunctive attempt takes nearly all of their default run and this change does not touch it.
# ROUNDS and FILES shorten a smoke run, OUTNAME names its directory, D the run's folder.
set -u
D=${D:-$HOME/lanes/horn-1025b}
F=$D/files
OUT=$D/out/${OUTNAME:-timing1025}; mkdir -p $OUT
{ echo "host=$(hostname) start=$(date -Is)"; uptime; free -g; sha256sum $D/bin/tip $D/bin/base; cat $D/bin/*.head; (cd $F && sha256sum *); } > $OUT/before.txt
files="${FILES:-ore_ont_4510.owl ore_ont_12901.owl ore_ont_14375.owl ore_ont_10951.owl ore_ont_14817.owl ore_ont_9663.owl go.ofn chebi_lite.ofn ore_ont_9855.owl ore_ont_8737.owl ore_ont_9654.owl ore_ont_4572.owl ore_ont_6132.owl ore_ont_10754.owl ore_ont_16744.owl ore_ont_10073.owl ore_ont_1370.owl}"
one() { # round arm name threads rep file [flags]
  local h=2 bin=tip
  case ${2%-horn} in l0) h=0;; l1) h=1;; base|base-el) bin=base;; esac
  mkdir -p $OUT/r$1/$2
  SPARKLES_HORN_RULES=$h systemd-run --user --scope -q -p MemoryMax=13G -p MemorySwapMax=0 -p OOMPolicy=continue \
    timeout 900 $D/bin/$bin "${@:7}" --threads $4 --repeat $5 --taxonomy $OUT/r$1/$2/$3.t$4.tax $6 > $OUT/r$1/$2/$3.t$4.json 2> $OUT/r$1/$2/$3.t$4.err
  echo "r$1 $2 $3 t$4 status=$? $(LC_ALL=C sort $OUT/r$1/$2/$3.t$4.tax | sha256sum | cut -c1-16)" >> $OUT/log.txt
  rm -f $OUT/r$1/$2/$3.t$4.tax
}
for round in ${ROUNDS:-1 2 3 4 5 6}; do
  order="base l0 l1 l2"; [ $((round % 2)) -eq 0 ] && order="l2 l1 l0 base"
  for f in $files; do
    n=${f%.*}
    for t in 1 4; do
      rep=5
      case $n in ore_ont_8737|ore_ont_9654) rep=2;; ore_ont_16744) rep=1;; esac
      case $n in
        ore_ont_14817|ore_ont_9663)
          for arm in $order; do one $round $arm-horn $n $t 3 $F/$f --dl-attempt-work 1; done;;
        *)
          for arm in $order; do one $round $arm $n $t $rep $F/$f; done;;
      esac
      case $n in ore_ont_4510|ore_ont_12901|ore_ont_14375|ore_ont_10951)
        els="base-el el"; [ $((round % 2)) -eq 0 ] && els="el base-el"
        for arm in $els; do one $round $arm $n $t $rep $F/$f --el-only; done;;
      esac
    done
  done
done
{ echo "end=$(date -Is)"; uptime; } > $OUT/after.txt
echo TIMING-DONE >> $OUT/log.txt
