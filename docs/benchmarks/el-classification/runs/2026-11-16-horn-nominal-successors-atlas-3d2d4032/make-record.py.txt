def rep(p):
    try: return json.loads([l for l in open(p) if l.startswith('{')][-1])
    except Exception: return None
def runs(root, pattern):
    out = []
    for p in sorted(glob.glob(os.path.join(root, pattern), recursive=True)):
        d = rep(p)
        out.append({'path': os.path.relpath(p, root), 'report': {k: d.get(k) for k in keep} if d else None})
    return out
