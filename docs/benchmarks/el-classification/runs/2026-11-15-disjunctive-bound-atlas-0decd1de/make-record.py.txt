# make-record.py LANE DEST: builds the benchmark record of item 990 from the atlas timing outputs,
# the laptop's distribution of the attempts' conclusions and the laptop's taxonomy sweep
import json, sys, os, glob
lane, dest = sys.argv[1], sys.argv[2]
keep = ['classifyMs', 'indexMs', 'saturationMs', 'dlTestingMs', 'dlTests', 'taxonomyMs', 'tier', 'hornOverBudget', 'dlOverBudget',
        'dlAttemptOver', 'dlWork', 'dlContexts', 'dlStored', 'outside', 'conclusions', 'contexts', 'classes',
        'peakEstimatedBytes', 'peakRssKib', 'threads']
def rep(p):
    try: return json.loads([l for l in open(p) if l.startswith('{')][-1])
    except Exception: return None
def runs(root, pattern):
    out = []
    for p in sorted(glob.glob(os.path.join(root, pattern), recursive=True)):
        d = rep(p)
        out.append({'path': os.path.relpath(p, root), 'report': {k: d.get(k) for k in keep} if d else None})
    return out
def text(p):
    return open(p).read() if os.path.exists(p) else None
def diag(root):
    out = {}
    for p in sorted(glob.glob(os.path.join(root, '*.err'))):
        n = os.path.basename(p)[:-4]
        out[n] = [l.rstrip('\n') for l in open(p) if l.startswith('D990')]
    return out
a = os.path.join(lane, 'atlas', 'timing')

               'scope': 'one process per run in a 13 GiB systemd scope without swap on atlas; on the laptop one 12 GiB scope for the distribution runs and one 14 GiB scope for three sweep processes at a time',

  'method': 'timing.sh ran three rounds of alternating order of base and tip, one process per side, file and thread count with --repeat 5, on the files whose attempt finishes, then three rounds of alternating order on four threads and one round on one thread of the files whose attempt passed its memory budget, with wall times and taxonomy hashes. dist.sh ran the diagnostic build of the base on the files that attempt the disjunctive rules on one and eight threads, and with --el-only on eight threads. gen.sh ran it on the generated ontologies on one thread with a 30 s timeout. ab.sh ran base and a build with the bound in steps (tip1) on eight threads on the laptop.
