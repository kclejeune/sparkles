#!/usr/bin/env bash
# corpus.sh: base and tip on GO, ChEBI lite, the 298 ORE 2015 EL files and the 207 DL
# development files, one and four threads. Per run: exit status, taxonomy hash, stats and
# a hash of the excluded axioms with their rules and messages, in out/corpus.txt, which
# ends with CORPUS-DONE.
set -u
L=/home/kclejeune/repositories/github.com/kclejeune/sparkles/.artifacts/worktrees/el-index-alloc/.artifacts/lanes/el-index-alloc
W=/home/kclejeune/repositories/github.com/kclejeune/sparkles/.artifacts/worktrees
EL=$W/el-956/.artifacts/lanes/el-956/data/el-files
D=$W/horn-cost-1025/.artifacts/lanes/horn-cost-1025/data
DEV=$W/horn-cost-693/.artifacts/lanes/horn-cost-693/data/dev
O=$L/out/corpus; mkdir -p $O
: > $L/out/corpus.txt
for f in $D/go.ofn $D/chebi_lite.ofn $(ls $EL/*) $(ls $DEV/*); do
  n=$(basename $f); n=${n%.*}
  for t in 1 4; do for b in base tip; do
    EL_CLASSIFY_EXCLUDED=1 systemd-run --user --scope -q -p MemoryMax=10G -p MemorySwapMax=1G -p OOMPolicy=continue \
      timeout 600 $L/bin/el_classify.$b --threads $t --taxonomy $O/$n.tax $f > $O/$n.json 2> $O/$n.err
    st=$?
    sha=$(LC_ALL=C sort $O/$n.tax 2>/dev/null | sha256sum | cut -c1-16); rm -f $O/$n.tax
    ex=$(grep '^excluded ' $O/$n.err | sha256sum | cut -c1-16); nex=$(grep -c '^excluded ' $O/$n.err)
    echo "$n $b t$t $st $sha $ex $nex $(grep '^{' $O/$n.json | jq -c '[.tier,.dlAttemptOver,.dlOverBudget,.hornOverBudget,.outside,.unsatisfiable,.consistency,.classifyMedianMs]')" >> $L/out/corpus.txt
  done; done
done
echo CORPUS-DONE >> $L/out/corpus.txt
