RUNDIR/record.json from the forge and atlas outputs, the laptop's instruction counts, the stage
probe and the corpus sweep, and copies the scripts beside it as .txt files."""
import glob, json, os, re, shutil, sys
L = os.path.dirname(os.path.abspath(__file__))
out = sys.argv[1]
os.makedirs(out, exist_ok=True)

def lines(p):
    return [json.loads(l) for l in open(p) if l.startswith("{")]

def processes(d):
    rows = []
    for p in sorted(glob.glob(f"{d}/timing/*.json")):
        rep = [json.loads(l) for l in open(p) if l.startswith("{")]
        err = open(p[:-5] + ".err").read() if os.path.exists(p[:-5] + ".err") else ""
        keep = ("classifyMs", "classifyMedianMs", "indexMs", "saturationMs", "taxonomyMs", "readMs",
                "tier", "outside", "conclusions", "contexts", "estimatedBytes", "peakEstimatedBytes",
                "peakRssKib", "peakRssKibAfterRead", "threads", "consistency", "unsatisfiable")
        rows.append({"path": os.path.basename(p),
                     "report": {k: rep[0].get(k) for k in keep} if rep else None, "stderr": err})
    return rows

def sums(p):
    return {l.split()[1]: l.split()[0] for l in open(p) if l.strip()}

def insn(prefix):
    res = {}
    for arm in ("base", "e21", "e22"):
        per = []
        for rnd in (1, 2):
            v = {}
            for rep in (1, 3):
                f = f"{L}/insn/{prefix}{arm}.r{rep}.{rnd}.csv"
                v[rep] = int([x for x in open(f) if "cpu_core" in x][0].split(",")[0])
            per.append({"repeat1": v[1], "repeat3": v[3], "per_classification": (v[3] - v[1]) // 2})
        res[{"e22": "tip"}.get(arm, arm)] = per
    return res

def probe(p):
    runs = []
    cur = {}
    for l in open(p):
        for k, v in re.findall(r"(\w+_us)=(\d+)", l):
            cur[k] = int(v)
        if "index_us" in l:
            runs.append(cur); cur = {}
    return runs
