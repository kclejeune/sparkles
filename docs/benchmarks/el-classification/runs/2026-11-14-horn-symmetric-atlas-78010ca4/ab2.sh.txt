#!/usr/bin/env bash
# ab2.sh: six rounds of rotating order on GO and ChEBI lite, sides base, lane (78010ca4) and
# final (2fb430aa), one and four threads, five classifications per process.
# The caller holds ~/lanes/.heavy.lock.
set -u
D=$HOME/lanes/horn-budget-962
OUT=$D/out/ab2
mkdir -p "$OUT"
{ echo "host=$(hostname) start=$(date -Is)"; uptime; free -g; sha256sum $D/bin/*; } > "$OUT/before.txt"
sides=(base lane final)
for round in 1 2 3 4 5 6; do
  k=$(( (round - 1) % 3 ))
  order="${sides[$k]} ${sides[$(( (k + 1) % 3 ))]} ${sides[$(( (k + 2) % 3 ))]}"
  for t in 1 4; do
    for f in go.ofn chebi_lite.ofn; do
      n=${f%.ofn}
      for tag in $order; do
        mkdir -p "$OUT/r$round/t$t/$tag"
        systemd-run --user --scope -q -p MemoryMax=13G -p MemorySwapMax=0 -p OOMPolicy=continue \
          timeout 600 "$D/bin/$tag" --threads $t --repeat 5 "$D/files/$f" > "$OUT/r$round/t$t/$tag/$n.json" 2> "$OUT/r$round/t$t/$tag/$n.err"
        echo "r$round t$t $tag $n status=$?" >> "$OUT/log.txt"
      done
    done
  done
done
{ echo "end=$(date -Is)"; uptime; } > "$OUT/after.txt"
echo AB2-DONE >> "$OUT/log.txt"
